def calculate_makespan(seq, matrix):
    """Calculate makespan for a job sequence on given machines.

    Args:
        seq: list of job indices (0-based) in processing order.
        matrix: 2D list of processing times (n_jobs x n_machines).

    Returns:
        float: makespan (completion time of last job on last machine).
    """
    n = len(seq)
    m = len(matrix[0])
    C = [[0.0] * m for _ in range(n)]
    for j in range(m):
        for i in range(n):
            job = seq[i]
            if j == 0:
                C[i][j] = matrix[job][j]
            else:
                prev = 0.0 if i == 0 else C[i - 1][j]
                C[i][j] = max(prev, C[i][j - 1]) + matrix[job][j]
    return C[n - 1][m - 1]
