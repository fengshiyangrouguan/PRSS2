def correct_greedy_crew_assignment(N: int, K: int, time_limit: float, tasks: dict, arcs: dict) -> dict:
    """Correct greedy crew assignment using next-fit style.

    Args:
        N: total number of tasks to cover
        K: number of crews
        time_limit: maximum duration any crew may cover
        tasks: dict mapping task_id to [start_time, end_time] float pair
        arcs: dict mapping (prev_task, next_task) -> connection_cost float

    Returns:
        {"crews": list of lists}, where each inner list is the ordered task_ids for one crew
        (empty lists if any task could not be assigned)
    """
    import sys
    # Sort tasks by start time
    task_list = sorted(tasks.keys(), key=lambda t: tasks[t][0])
    crews = [[] for _ in range(K)]
    crew_end = [0.0] * K
    assigned = set()
    for task in task_list:
        if task in assigned:
            continue
        best_crew = -1
        best_prev = None
        for c_idx in range(K):
            crew = crews[c_idx]
            if not crew:
                # start new crew
                s, f = tasks[task]
                if f - s > time_limit:
                    continue
                best_crew = c_idx
                best_prev = None
                break  # take first possible empty crew
            last = crew[-1]
            if (last, task) not in arcs:
                continue
            cost = arcs[(last, task)]
            last_f = crew_end[c_idx]
            if last_f > tasks[task][0]:
                continue
            new_f = max(last_f, tasks[task][1])
            if new_f - tasks[crew[0]][0] > time_limit:
                continue
            best_crew = c_idx
            best_prev = last
            break  # take first feasible crew (next-fit)
        if best_crew != -1:
            if best_prev is None:
                crews[best_crew] = [task]
