            if isinstance(o, str): txts.append(o)

                for v in o.values(): walk(v)
            elif isinstance(o, list):
                for v in o: walk(v)

        blob = "\n".join(txts)
        for n in names:
            if n in blob:
                out[n].write(blob + "\n===8<===\n")
