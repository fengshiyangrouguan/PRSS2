def pack_stocks(stocks: list, pieces: list) -> dict:
    """Pack pieces into stocks using first-fit with orientation, respecting min/max counts.

    Args:
        stocks: list of dicts with 'length' and 'width' (stock_type defaults to 1).
        pieces: list of dicts with 'length', 'width', 'min', 'max'.

    Returns:
        dict: placements where key is instance_id, value has 'stock_type' and 'placements' list.
    """
    if not stocks or not pieces:
        return {}
    placements = {}
    current_instance_id = 1
    stock_type = 1
    # Group pieces by type
    from collections import defaultdict
    type_pieces = defaultdict(list)
    for p in pieces:
        type_pieces[p['type']].append(p)
    for piece_type, piece_list in type_pieces.items():
        min_count = piece_list[0]['min']
        max_count = piece_list[0]['max']
        if min_count == 0:
            continue
        # Find suitable stock
        stock_idx = None
        for i, s in enumerate(stocks):
            s_l = s['length']
            s_w = s['width']
            for p in piece_list:
                p_l0 = p['length']
                p_w0 = p['width']
                if (p_l0 <= s_l and p_w0 <= s_w) or (p_w0 <= s_l and p_l0 <= s_w):
                    stock_idx = i
                    break
            if stock_idx is not None:
                break
        if stock_idx is None:
            continue
        # Allocate min_count
        for _ in range(min_count):
            s_l = stocks[stock_idx]['length']
            s_w = stocks[stock_idx]['width']
            for p in piece_list:
                p_l0 = p['length']
                p_w0 = p['width']
                if p_l0 <= s_l and p_w0 <= s_w:
                    orient = 0
                    pl = p_l0
                    pw = p_w0
                    break
                elif p_w0 <= s_l and p_l0 <= s_w:
                    orient = 1
                    pl = p_w0
                    pw = p_l0
                    break
            else:
                continue
            placements[current_instance_id] = {
                "stock_type": stock_type,
                "placements": [{
                    "piece": piece_type,
                    "x": 0.0,
                    "y": 0.0,
                    "orientation": orient
                }]
            }
            current_instance_id += 1
    return placements
